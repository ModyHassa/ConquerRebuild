cmake_minimum_required(VERSION 3.15)
project(ConquerRebuild CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(conquer_server src/main.cpp)
if(WIN32)
    target_link_libraries(conquer_server ws2_32)
else()
    find_package(Threads REQUIRED)
    target_link_libraries(conquer_server Threads::Threads)
endif()
